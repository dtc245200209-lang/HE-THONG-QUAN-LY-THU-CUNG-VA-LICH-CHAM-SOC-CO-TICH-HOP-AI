% CẤU HÌNH TYPOGRAPHY CHUẨN HỌC THUẬT CHO LATEX
% Sử dụng hệ font Times New Roman chuẩn mực của văn bản học thuật ICTU,
% tương thích hoàn toàn với XeLaTeX và hỗ trợ tiếng Việt Unicode.
% ==============================================================================

% Font văn bản chính (Roman / Serif)
\setmainfont{Times New Roman}[
    Ligatures = TeX
]

% Font không chân (Sans Serif)
\setsansfont{Arial}[
    Ligatures = TeX
]

% Font đơn khoảng cách (Monospace) cho mã nguồn và URL
\setmonofont{Courier New}[
    Scale = 0.9,
    Ligatures = TeX
]
